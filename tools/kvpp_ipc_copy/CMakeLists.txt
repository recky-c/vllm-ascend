cmake_minimum_required(VERSION 3.16)
project(a3_kv_transfer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED ASCEND_CANN_PACKAGE_PATH OR NOT DEFINED SOC_VERSION)
  message(FATAL_ERROR "Pass the measured CANN path and SOC_VERSION")
endif()
set(RUN_MODE npu)
include(${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(kv_copy SHARED copy.cpp)
add_executable(kv_probe probe.cpp)
target_include_directories(kv_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/include)
target_link_directories(kv_probe PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(kv_probe PRIVATE ascendcl dl pthread)
target_compile_options(kv_probe PRIVATE -O2 -Wall -Wextra)
